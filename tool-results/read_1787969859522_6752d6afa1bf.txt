     1→"use client";
     2→import { useEffect, useState, useCallback, useMemo, Suspense } from "react";
     3→import { useRouter, useSearchParams } from "next/navigation";
     4→import { AppShell } from "@/components/shared/AppShell";
     5→import { EmptyState } from "@/components/shared/EmptyState";
     6→import { LoadingSkeleton } from "@/components/shared/LoadingSkeleton";
     7→import { JobCard, type WorkerJobCardData } from "@/components/worker/JobCard";
     8→import { Card, CardContent } from "@/components/ui/card";
     9→import { Button } from "@/components/ui/button";
    10→import { Input } from "@/components/ui/input";
    11→import { Label } from "@/components/ui/label";
    12→import { Badge } from "@/components/ui/badge";
    13→import { Switch } from "@/components/ui/switch";
    14→import {
    15→  Select, SelectTrigger, SelectValue, SelectContent, SelectItem,
    16→} from "@/components/ui/select";
    17→import { useLanguage } from "@/lib/i18n/LanguageProvider";
    18→import { useSavedJobs } from "@/hooks/use-saved-jobs";
    19→import { useSession } from "next-auth/react";
    20→import { motion } from "framer-motion";
    21→import {
    22→  Search, X, Zap, Bookmark, ArrowDownWideNarrow, MapPin, Briefcase,
    23→  ChevronDown, Loader2, Compass, SlidersHorizontal, Star, Radio,
    24→} from "lucide-react";
    25→import type { Skill } from "@/lib/schemas";
    26→import { toast } from "sonner";
    27→
    28→const TRADE_NAMES = new Set(["Electrician", "Plumber", "Welder", "CNC Operator", "Fitter", "Delivery Executive", "Carpenter", "Mason"]);
    29→
    30→interface FeedResponse {
    31→  items: WorkerJobCardData[];
    32→  total: number;
    33→  page: number;
    34→  pageSize: number;
    35→  hasNext: boolean;
    36→}
    37→
    38→type SortKey = "match" | "wage" | "newest" | "nearest";
    39→
    40→const PAGE_SIZE = 9;
    41→
    42→function JobBoard() {
    43→  const { t } = useLanguage();
    44→  const router = useRouter();
    45→  const search = useSearchParams();
    46→  const { data: session, status } = useSession();
    47→  const { savedIds, savedCount } = useSavedJobs();
    48→
    49→  // ---- URL-synced state (shareable deep links: /jobs?q=welder&city=...&sort=wage) ----
    50→  const [query, setQuery] = useState(search.get("q") ?? "");
    51→  const [tradeId, setTradeId] = useState(search.get("trade") ?? "");
    52→  const [city, setCity] = useState(search.get("city") ?? "");
    53→  const [shift, setShift] = useState<"day" | "night" | "any">((search.get("shift") as any) ?? "any");
    54→  const [sort, setSort] = useState<SortKey>((search.get("sort") as SortKey) ?? "match");
    55→  const [urgentOnly, setUrgentOnly] = useState(search.get("urgent") === "1");
    56→  const [savedOnly, setSavedOnly] = useState(search.get("saved") === "1");
    57→  const [topEmployersOnly, setTopEmployersOnly] = useState(search.get("top") === "1");
    58→
    59→  // ---- data ----
    60→  const [skills, setSkills] = useState<Skill[]>([]);
    61→  const [jobs, setJobs] = useState<WorkerJobCardData[] | null>(null);
    62→  const [hasNext, setHasNext] = useState(false);
    63→  const [loadingMore, setLoadingMore] = useState(false);
    64→  const [profileExists, setProfileExists] = useState<boolean | null>(null);
    65→  const [appliedIds, setAppliedIds] = useState<Set<string>>(new Set());
    66→
    67→  // ---- worker "Available today" quick-toggle (round 11) ----
    68→  // Visible only to logged-in workers. Mirrors the toggle on /home but lets
    69→  // a worker flip their availability while browsing the board.
    70→  const [availableToday, setAvailableToday] = useState<boolean | null>(null);
    71→  const [availableSaving, setAvailableSaving] = useState(false);
    72→
    73→  // Skills taxonomy (for trade filter)
    74→  useEffect(() => {
    75→    fetch("/api/skills")
    76→      .then(r => r.json())
    77→      .then((d: { items: Skill[] }) => setSkills(d.items ?? []))
    78→      .catch(() => setSkills([]));
    79→  }, []);
    80→
    81→  // Worker profile existence (redirect to onboarding when missing).
    82→  // Non-workers (employer/admin) can browse the board too — no profile needed.
    83→  useEffect(() => {
    84→    if (status !== "authenticated") return;
    85→    const role = (session?.user as { role?: string } | undefined)?.role;
    86→    if (role && role !== "worker") { setProfileExists(true); return; }
    87→    fetch("/api/worker/profile", { cache: "no-store" })
    88→      .then(r => {
    89→        if (r.status === 403 || r.status === 404) { setProfileExists(false); return null; }
    90→        if (!r.ok) return null;
    91→        return r.json();
    92→      })
    93→      .then(data => {
    94→        if (data === null) return;
    95→        setProfileExists(true);
    96→        // Round 11: seed the quick-toggle from the worker's stored flag.
    97→        if (typeof data.availableToday === "boolean") {
    98→          setAvailableToday(data.availableToday);
    99→        }
   100→      })
   101→      .catch(() => setProfileExists(false));
   102→  }, [status, session]);
   103→
   104→  // Round 11: workers can flip availability directly from the board header.
   105→  async function toggleAvailableToday(checked: boolean) {
   106→    if (availableToday === null) return;
   107→    setAvailableToday(checked);
   108→    setAvailableSaving(true);
   109→    try {
   110→      const res = await fetch("/api/worker/profile", {
   111→        method: "PATCH",
   112→        headers: { "content-type": "application/json" },
   113→        body: JSON.stringify({ availableToday: checked }),
   114→      });
   115→      if (!res.ok) throw new Error("patch-failed");
   116→      toast.success(checked ? t("boardAvailableTodayOn") : t("boardAvailableTodayOff"));
   117→    } catch {
   118→      setAvailableToday(!checked); // rollback
   119→      toast.error(t("errGeneric"));
   120→    } finally {
   121→      setAvailableSaving(false);
   122→    }
   123→  }
   124→
   125→  useEffect(() => {
   126→    if (profileExists === false && status === "authenticated") {
   127→      const role = (session?.user as any)?.role;
   128→      if (role === "worker") router.replace("/onboarding/worker");
   129→    }
   130→  }, [profileExists, status, session, router]);
   131→
   132→  // Applied jobs (mark cards so the Apply button doesn't double-submit)
   133→  useEffect(() => {
   134→    if (status !== "authenticated") return;
   135→    fetch("/api/applications/mine", { cache: "no-store" })
   136→      .then(r => (r.ok ? r.json() : { items: [] }))
   137→      .then((d: { items: { jobId: string }[] }) => {
   138→        setAppliedIds(new Set((d.items ?? []).map(a => a.jobId)));
   139→      })
   140→      .catch(() => {});
   141→  }, [status]);
   142→
   143→  // Server-side query (trade/shift/urgent). Browse-all: distanceKm=200 overrides
   144→  // the worker-radius filter so every open job on the platform is reachable here.
   145→  const serverQuery = useMemo(() => {
   146→    const p = new URLSearchParams();
   147→    p.set("pageSize", String(PAGE_SIZE));
   148→    p.set("distanceKm", "200");
   149→    if (tradeId) p.set("tradeId", tradeId);
   150→    if (shift !== "any") p.set("shift", shift);
   151→    if (urgentOnly) p.set("urgentOnly", "true");
   152→    return p.toString();
   153→  }, [tradeId, shift, urgentOnly]);
   154→
   155→  // Fetch page 1 (debounced) whenever server-side filters change
   156→  const loadFirstPage = useCallback(async () => {
   157→    setJobs(null);
   158→    try {
   159→      const res = await fetch(`/api/jobs?${serverQuery}&page=1`, { cache: "no-store" });
   160→      if (!res.ok) { setJobs([]); setHasNext(false); return; }
   161→      const data = (await res.json()) as FeedResponse;
   162→      setJobs(data.items ?? []);
   163→      setHasNext(!!data.hasNext);
   164→    } catch {
   165→      setJobs([]);
   166→      setHasNext(false);
   167→    }
   168→  }, [serverQuery]);
   169→
   170→  useEffect(() => {
   171→    if (!profileExists) return;
   172→    const id = setTimeout(loadFirstPage, 250);
   173→    return () => clearTimeout(id);
   174→  }, [loadFirstPage, profileExists]);
   175→
   176→  // Load-more appends the next page
   177→  const loadMore = useCallback(async () => {
   178→    if (loadingMore || !jobs) return;
   179→    setLoadingMore(true);
   180→    try {
   181→      // Infer next page from the items we already hold
   182→      const nextPage = Math.floor(jobs.length / PAGE_SIZE) + 1;
   183→      const res = await fetch(`/api/jobs?${serverQuery}&page=${nextPage}`, { cache: "no-store" });
   184→      if (!res.ok) { toast.error(t("errGeneric")); return; }
   185→      const data = (await res.json()) as FeedResponse;
   186→      setJobs(prev => {
   187→        const seen = new Set((prev ?? []).map(j => j.id));
   188→        const fresh = (data.items ?? []).filter(j => !seen.has(j.id));
   189→        return [...(prev ?? []), ...fresh];
   190→      });
   191→      setHasNext(!!data.hasNext);
   192→    } catch {
   193→      toast.error(t("errGeneric"));
   194→    } finally {
   195→      setLoadingMore(false);
   196→    }
   197→  }, [loadingMore, jobs, serverQuery, t]);
   198→
   199→  // ---- URL sync (replace, no scroll jump) ----
   200→  useEffect(() => {
   201→    if (typeof window === "undefined") return;
   202→    const p = new URLSearchParams();
   203→    if (query.trim()) p.set("q", query.trim());
   204→    if (tradeId) p.set("trade", tradeId);
   205→    if (city) p.set("city", city);
   206→    if (shift !== "any") p.set("shift", shift);
   207→    if (sort !== "match") p.set("sort", sort);
   208→    if (urgentOnly) p.set("urgent", "1");
   209→    if (savedOnly) p.set("saved", "1");
   210→    if (topEmployersOnly) p.set("top", "1");
   211→    const qs = p.toString();
   212→    const target = qs ? `/jobs?${qs}` : "/jobs";
   213→    if (window.location.pathname + window.location.search !== target) {
   214→      router.replace(target, { scroll: false });
   215→    }
   216→  }, [query, tradeId, city, shift, sort, urgentOnly, savedOnly, topEmployersOnly, router]);
   217→
   218→  const tradeSkills = useMemo(() => skills.filter(s => TRADE_NAMES.has(s.nameEn)), [skills]);
   219→  const tradeName = useMemo(
   220→    () => tradeSkills.find(s => s.id === tradeId)?.nameEn ?? null,
   221→    [tradeSkills, tradeId],
   222→  );
   223→
   224→  // Cities derived from loaded jobs (only cities that actually have jobs)
   225→  const cities = useMemo(() => {
   226→    if (!jobs) return [];
   227→    return Array.from(new Set(jobs.map(j => j.city))).sort();
   228→  }, [jobs]);
   229→
   230→  // ---- client-side pipeline: text search → city → saved → top employers → sort ----
   231→  const visible = useMemo(() => {
   232→    if (!jobs) return null;
   233→    const q = query.trim().toLowerCase();
   234→    let out = jobs.filter(j => {
   235→      if (q) {
   236→        const hay = [
   237→          j.title,
   238→          j.employer?.companyName ?? "",
   239→          j.trade?.nameEn ?? "",
   240→          j.city,
   241→          ...j.skills.map(s => s.skill?.nameEn ?? ""),
   242→        ].join(" ").toLowerCase();
   243→        if (!hay.includes(q)) return false;
   244→      }
   245→      if (city && j.city !== city) return false;
   246→      if (savedOnly && !savedIds.has(j.id)) return false;
   247→      // Round 8: "Top employers" — jobs from employers rated ≥4.5 by 3+ workers
   248→      if (topEmployersOnly) {
   249→        const r = j.employer;
   250→        if (!r?.ratingAvg || !r.ratingCount || r.ratingAvg < 4.5 || r.ratingCount < 3) return false;
   251→      }
   252→      return true;
   253→    });
   254→    out = [...out].sort((a, b) => {
   255→      if (sort === "match") return (b.matchScore ?? -1) - (a.matchScore ?? -1);
   256→      if (sort === "wage") return b.wageMax - a.wageMax;
   257→      if (sort === "nearest") return (a.distanceKm ?? Infinity) - (b.distanceKm ?? Infinity);
   258→      return +new Date(b.createdAt) - +new Date(a.createdAt);
   259→    });
   260→    return out;
   261→  }, [jobs, query, city, savedOnly, savedIds, sort, topEmployersOnly]);
   262→
   263→  // ---- active filter chips ----
   264→  interface Chip { key: string; label: string; clear: () => void }
   265→  const chips: Chip[] = useMemo(() => {
   266→    const out: Chip[] = [];
   267→    if (query.trim()) out.push({ key: "q", label: `“${query.trim()}”`, clear: () => setQuery("") });
   268→    if (tradeName) out.push({ key: "trade", label: tradeName, clear: () => setTradeId("") });
   269→    if (city) out.push({ key: "city", label: city, clear: () => setCity("") });
   270→    if (shift !== "any") out.push({ key: "shift", label: shift === "day" ? "Day" : "Night", clear: () => setShift("any") });
   271→    if (urgentOnly) out.push({ key: "urgent", label: t("feedUrgent"), clear: () => setUrgentOnly(false) });
   272→    if (savedOnly) out.push({ key: "saved", label: t("boardSavedOnly"), clear: () => setSavedOnly(false) });
   273→    if (topEmployersOnly) out.push({ key: "top", label: t("boardTopEmployers"), clear: () => setTopEmployersOnly(false) });
   274→    return out;
   275→  }, [query, tradeName, city, shift, urgentOnly, savedOnly, topEmployersOnly, t]);
   276→
   277→  function clearAll() {
   278→    setQuery("");
   279→    setTradeId("");
   280→    setCity("");
   281→    setShift("any");
   282→    setUrgentOnly(false);
   283→    setSavedOnly(false);
   284→    setTopEmployersOnly(false);
   285→    // keep sort — it's a view preference, not a filter
   286→  }
   287→
   288→  const hasActiveFilters = chips.length > 0;
   289→  const isWorker = (session?.user as { role?: string } | undefined)?.role === "worker";
   290→
   291→  if (status === "loading" || profileExists === null) {
   292→    return (
   293→      <AppShell>
   294→        <LoadingSkeleton count={3} />
   295→      </AppShell>
   296→    );
   297→  }
   298→
   299→  return (
   300→    <AppShell>
   301→      <div className="flex flex-col gap-4">
   302→        {/* ---------- Hero / search header ---------- */}
   303→        <section className="relative overflow-hidden rounded-2xl border border-border bg-gradient-to-br from-primary/10 via-card to-accent/10 p-4 sm:p-6">
   304→          {/* decorative bridge arcs */}
   305→          <div aria-hidden className="pointer-events-none absolute -right-10 -top-16 size-48 rounded-full border-[14px] border-primary/10" />
   306→          <div aria-hidden className="pointer-events-none absolute -right-4 -top-10 size-28 rounded-full border-[10px] border-accent/15" />
   307→
   308→          <div className="relative flex flex-col gap-3">
   309→            <div className="flex items-start justify-between gap-3 flex-wrap">
   310→              <div>
   311→                <h1 className="text-2xl font-bold tracking-tight flex items-center gap-2">
   312→                  <Compass className="size-6 text-primary" />
   313→                  {t("boardTitle")}
   314→                </h1>
   315→                <p className="text-sm text-muted-foreground mt-1">{t("boardSubtitle")}</p>
   316→              </div>
   317→              {visible && (
   318→                <Badge variant="secondary" className="tabular-nums px-3 py-1 text-xs shrink-0">
   319→                  {t(visible.length === 1 ? "boardResultOne" : "boardResults", { count: visible.length })}
   320→                </Badge>
   321→              )}
   322→            </div>
   323→
   324→            {/* Search bar */}
   325→            <div className="relative">
   326→              <Search className="absolute left-3.5 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" aria-hidden />
   327→              <Input
   328→                type="search"
   329→                value={query}
   330→                onChange={e => setQuery(e.target.value)}
   331→                placeholder={t("boardSearchPlaceholder")}
   332→                aria-label={t("boardSearchPlaceholder")}
   333→                className="pl-10 pr-10 h-12 text-base rounded-xl bg-card shadow-sm border-border/80 focus-visible:ring-2"
   334→              />
   335→              {query && (
   336→                <button
   337→                  type="button"
   338→                  onClick={() => setQuery("")}
   339→                  aria-label={t("boardClearAll")}
   340→                  className="absolute right-2.5 top-1/2 -translate-y-1/2 grid place-items-center size-7 rounded-full text-muted-foreground hover:bg-accent hover:text-foreground transition-colors"
   341→                >
   342→                  <X className="size-4" />
   343→                </button>
   344→              )}
   345→            </div>
   346→
   347→            {/* Round 11: worker "Available today" quick toggle */}
   348→            {isWorker && availableToday !== null && (
   349→              <motion.button
   350→                type="button"
   351→                onClick={() => toggleAvailableToday(!availableToday)}
   352→                disabled={availableSaving}
   353→                initial={{ opacity: 0, y: -4 }}
   354→                animate={{ opacity: 1, y: 0 }}
   355→                transition={{ duration: 0.25, ease: "easeOut" }}
   356→                aria-pressed={availableToday}
   357→                aria-label={t("boardAvailableToday")}
   358→                className={`group inline-flex items-center gap-3 self-start rounded-xl border px-4 py-2.5 text-left transition-all duration-200 ${
   359→                  availableToday
   360→                    ? "border-emerald-400/70 bg-emerald-50 dark:bg-emerald-950/40 shadow-[0_0_0_3px_rgba(16,185,129,0.08)]"
   361→                    : "border-border bg-card hover:border-emerald-300 hover:bg-emerald-50/40 dark:hover:bg-emerald-950/20"
   362→                }`}
   363→              >
   364→                <span
   365→                  aria-hidden
   366→                  className={`relative grid size-9 place-items-center rounded-full transition-colors ${
   367→                    availableToday
   368→                      ? "bg-emerald-500 text-white"
   369→                      : "bg-muted text-muted-foreground group-hover:bg-emerald-100 group-hover:text-emerald-700"
   370→                  }`}
   371→                >
   372→                  {availableSaving ? (
   373→                    <Loader2 className="size-4 animate-spin" />
   374→                  ) : (
   375→                    <Radio className={`size-4 ${availableToday ? "animate-pulse" : ""}`} />
   376→                  )}
   377→                  {availableToday && (
   378→                    <span aria-hidden className="absolute inset-0 rounded-full bg-emerald-400 opacity-40 motion-safe:animate-ping" />
   379→                  )}
   380→                </span>
   381→                <span className="flex flex-col gap-0.5 min-w-0">
   382→                  <span className={`text-sm font-semibold leading-tight ${availableToday ? "text-emerald-800 dark:text-emerald-200" : "text-foreground"}`}>
   383→                    {availableToday ? t("boardAvailableTodayOn") : t("boardAvailableTodayOff")}
   384→                  </span>
   385→                  <span className="text-[11px] text-muted-foreground truncate">
   386→                    {t("boardAvailableTodayHint")}
   387→                  </span>
   388→                </span>
   389→                <Switch
   390→                  checked={availableToday}
   391→                  disabled={availableSaving}
   392→                  onCheckedChange={toggleAvailableToday}
   393→                  onClick={e => e.stopPropagation()} // avoid double-trigger from outer button
   394→                  className="ml-1 data-[state=checked]:bg-emerald-600"
   395→                  aria-label={t("boardAvailableToday")}
   396→                />
   397→              </motion.button>
   398→            )}
   399→          </div>
   400→        </section>
   401→
   402→        {/* ---------- Sticky toolbar ---------- */}
   403→        <div className="sticky top-14 z-20 -mx-4 sm:-mx-6 px-4 sm:px-6 py-2 bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/75 border-b border-border">
   404→          <div className="flex items-center gap-2 flex-wrap">
   405→            <SlidersHorizontal className="size-4 text-muted-foreground shrink-0" aria-hidden />
   406→
   407→            <Select value={tradeId || "any"} onValueChange={v => setTradeId(v === "any" ? "" : v)}>
   408→              <SelectTrigger className="h-9 w-auto min-w-28 gap-1 text-xs font-medium" aria-label={t("feedFilterTrade")}>
   409→                <Briefcase className="size-3.5 text-muted-foreground" />
   410→                <SelectValue />
   411→              </SelectTrigger>
   412→              <SelectContent>
   413→                <SelectItem value="any">{t("feedFilterTrade")}: any</SelectItem>
   414→                {tradeSkills.map(s => (
   415→                  <SelectItem key={s.id} value={s.id}>{s.nameEn}</SelectItem>
   416→                ))}
   417→              </SelectContent>
   418→            </Select>
   419→
   420→            <Select value={city || "any"} onValueChange={v => setCity(v === "any" ? "" : v)}>
   421→              <SelectTrigger className="h-9 w-auto min-w-24 gap-1 text-xs font-medium" aria-label={t("boardCity")}>
   422→                <MapPin className="size-3.5 text-muted-foreground" />
   423→                <SelectValue />
   424→              </SelectTrigger>
   425→              <SelectContent>
   426→                <SelectItem value="any">{t("boardAnyCity")}</SelectItem>
   427→                {cities.map(c => (
   428→                  <SelectItem key={c} value={c}>{c}</SelectItem>
   429→                ))}
   430→              </SelectContent>
   431→            </Select>
   432→
   433→            <Select value={shift} onValueChange={v => setShift(v as typeof shift)}>
   434→              <SelectTrigger className="h-9 w-auto min-w-24 text-xs font-medium" aria-label={t("feedFilterShift")}>
   435→                <SelectValue />
   436→              </SelectTrigger>
   437→              <SelectContent>
   438→                <SelectItem value="any">{t("feedFilterShift")}: any</SelectItem>
   439→                <SelectItem value="day">Day</SelectItem>
   440→                <SelectItem value="night">Night</SelectItem>
   441→              </SelectContent>
   442→            </Select>
   443→
   444→            <Select value={sort} onValueChange={v => setSort(v as SortKey)}>
   445→              <SelectTrigger className="h-9 w-auto min-w-32 gap-1 text-xs font-medium sm:ml-auto" aria-label={t("boardSortLabel")}>
   446→                <ArrowDownWideNarrow className="size-3.5 text-muted-foreground" />
   447→                <SelectValue />
   448→              </SelectTrigger>
   449→              <SelectContent>
   450→                <SelectItem value="match">{t("boardSortMatch")}</SelectItem>
   451→                <SelectItem value="wage">{t("boardSortWage")}</SelectItem>
   452→                <SelectItem value="nearest">{t("boardSortNearest")}</SelectItem>
   453→                <SelectItem value="newest">{t("boardSortNewest")}</SelectItem>
   454→              </SelectContent>
   455→            </Select>
   456→          </div>
   457→
   458→          {/* Toggle chips row */}
   459→          <div className="flex items-center gap-2 flex-wrap mt-2">
   460→            <label
   461→              className={`inline-flex items-center gap-2 h-9 rounded-full border px-3 cursor-pointer transition-colors ${urgentOnly ? "border-accent bg-accent/15 text-accent-foreground" : "border-border bg-card text-muted-foreground hover:border-accent/50"}`}
   462→            >
   463→              <Zap className="size-3.5" />
   464→              <span className="text-xs font-medium">{t("feedUrgent")}</span>
   465→              <Switch checked={urgentOnly} onCheckedChange={setUrgentOnly} className="scale-75 data-[state=checked]:bg-accent-foreground" aria-label={t("feedUrgent")} />
   466→            </label>
   467→
   468→            <label
   469→              className={`inline-flex items-center gap-2 h-9 rounded-full border px-3 cursor-pointer transition-colors ${savedOnly ? "border-primary bg-primary/10 text-primary" : "border-border bg-card text-muted-foreground hover:border-primary/50"}`}
   470→            >
   471→              <Bookmark className={`size-3.5 ${savedOnly ? "fill-primary" : ""}`} />
   472→              <span className="text-xs font-medium">{t("boardSavedOnly")}</span>
   473→              {savedCount > 0 && (
   474→                <Badge variant="secondary" className="text-[10px] tabular-nums h-5 px-1.5">{savedCount}</Badge>
   475→              )}
   476→              <Switch checked={savedOnly} onCheckedChange={setSavedOnly} className="scale-75" aria-label={t("boardSavedOnly")} />
   477→            </label>
   478→
   479→            <label
   480→              className={`inline-flex items-center gap-2 h-9 rounded-full border px-3 cursor-pointer transition-colors ${topEmployersOnly ? "border-amber-400 bg-amber-100 text-amber-800 dark:bg-amber-950/50 dark:text-amber-300" : "border-border bg-card text-muted-foreground hover:border-amber-400/60"}`}
   481→            >
   482→              <Star className={`size-3.5 ${topEmployersOnly ? "fill-amber-500 text-amber-500" : ""}`} />
   483→              <span className="text-xs font-medium">{t("boardTopEmployers")}</span>
   484→              <Switch checked={topEmployersOnly} onCheckedChange={setTopEmployersOnly} className="scale-75 data-[state=checked]:bg-amber-500" aria-label={t("boardTopEmployers")} />
   485→            </label>
   486→
   487→            {hasActiveFilters && (
   488→              <Button
   489→                type="button"
   490→                variant="ghost"
   491→                size="sm"
   492→                onClick={clearAll}
   493→                className="h-9 rounded-full text-xs gap-1 ml-auto text-muted-foreground hover:text-foreground"
   494→              >
   495→                <X className="size-3.5" />
   496→                {t("boardClearAll")}
   497→              </Button>
   498→            )}
   499→          </div>
   500→        </div>
   501→
   502→        {/* ---------- Active filter chips ---------- */}
   503→        {hasActiveFilters && (
   504→          <div className="flex items-center gap-2 flex-wrap" aria-label={t("boardClearAll")}>
   505→            {chips.map(c => (
   506→              <button
   507→                key={c.key}
   508→                type="button"
   509→                onClick={c.clear}
   510→                className="inline-flex items-center gap-1.5 h-8 rounded-full bg-primary/10 text-primary pl-3 pr-2 text-xs font-medium hover:bg-primary/15 transition-colors group"
   511→                aria-label={`${c.label} — remove filter`}
   512→              >
   513→                {c.label}
   514→                <X className="size-3.5 opacity-60 group-hover:opacity-100" />
   515→              </button>
   516→            ))}
   517→          </div>
   518→        )}
   519→
   520→        {/* ---------- Results ---------- */}
   521→        {jobs === null && <LoadingSkeleton count={4} />}
   522→
   523→        {jobs !== null && visible !== null && visible.length === 0 && query.trim() !== "" && (
   524→          <EmptyState
   525→            icon={Search}
   526→            title={t("boardSearchEmpty")}
   527→            description={t("boardSearchEmptyHint")}
   528→            action={
   529→              <Button type="button" variant="outline" size="sm" onClick={() => { setQuery(""); setCity(""); }}>
   530→                {t("boardClearAll")}
   531→              </Button>
   532→            }
   533→          />
   534→        )}
   535→
   536→        {jobs !== null && visible !== null && visible.length === 0 && query.trim() === "" && savedOnly && (
   537→          <EmptyState
   538→            icon={Bookmark}
   539→            title={t("boardNoSaved")}
   540→            description={t("boardNoSavedHint")}
   541→            action={
   542→              <Button type="button" variant="outline" size="sm" onClick={() => setSavedOnly(false)}>
   543→                {t("boardClearAll")}
   544→              </Button>
   545→            }
   546→          />
   547→        )}
   548→
   549→        {jobs !== null && visible !== null && visible.length === 0 && query.trim() === "" && !savedOnly && (
   550→          <EmptyState
   551→            icon={Search}
   552→            title={t("feedEmpty")}
   553→            description={t("boardSearchEmptyHint")}
   554→            action={
   555→              <Button type="button" variant="outline" size="sm" onClick={clearAll}>
   556→                {t("boardClearAll")}
   557→              </Button>
   558→            }
   559→          />
   560→        )}
   561→
   562→        {visible !== null && visible.length > 0 && (
   563→          <>
   564→            <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 xl:grid-cols-3">
   565→              {visible.map((job, i) => (
   566→                <motion.div
   567→                  key={job.id}
   568→                  initial={{ opacity: 0, y: 12 }}
   569→                  animate={{ opacity: 1, y: 0 }}
   570→                  transition={{ duration: 0.25, delay: Math.min(i, 6) * 0.05, ease: "easeOut" }}
   571→                >
   572→                  <JobCard job={job} applied={appliedIds.has(job.id)} />
   573→                </motion.div>
   574→              ))}
   575→            </div>
   576→
   577→            {/* Footer: count + load more / end of list */}
   578→            <div className="flex flex-col items-center gap-3 mt-2 pb-2">
   579→              <p className="text-xs text-muted-foreground tabular-nums">
   580→                {t("boardShowing", { shown: visible.length, total: jobs.length })}
   581→              </p>
   582→              {hasNext ? (
   583→                <Button type="button" variant="outline" onClick={loadMore} disabled={loadingMore} className="gap-2 min-h-11 px-6">
   584→                  {loadingMore
   585→                    ? <Loader2 className="size-4 animate-spin" />
   586→                    : <ChevronDown className="size-4" />}
   587→                  {t("boardLoadMore")}
   588→                </Button>
   589→              ) : (
   590→                <div className="flex items-center gap-3 w-full max-w-xs text-muted-foreground" aria-hidden>
   591→                  <span className="h-px flex-1 bg-border" />
   592→                  <span className="text-[10px] uppercase tracking-widest whitespace-nowrap">{t("boardEndOfList")}</span>
   593→                  <span className="h-px flex-1 bg-border" />
   594→                </div>
   595→              )}
   596→            </div>
   597→          </>
   598→        )}
   599→
   600→        {/* Screen-reader summary of current filters */}
   601→        <Card className="sr-only">
   602→          <CardContent>
   603→            {t(visible?.length === 1 ? "boardResultOne" : "boardResults", { count: visible?.length ?? 0 })}
   604→            {hasActiveFilters ? ` · ${chips.map(c => c.label).join(", ")}` : ""}
   605→          </CardContent>
   606→        </Card>
   607→      </div>
   608→    </AppShell>
   609→  );
   610→}
