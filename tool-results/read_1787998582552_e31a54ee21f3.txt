    84→export default function WorkerProfilePage() {
    85→  const { t, lang } = useLanguage();
    86→  const [profile, setProfile] = useState<WorkerProfileData | null>(null);
    87→  const [notFound, setNotFound] = useState(false);
    88→  const [saving, setSaving] = useState(false);
    89→  const [edit, setEdit] = useState({
    90→    fullName: "",
    91→    yearsExp: "0",
    92→    city: "",
    93→    wageMin: "0",
    94→    wageMax: "0",
    95→    shiftPref: "any" as "day" | "night" | "any",
    96→    bio: "",
    97→    photoUrl: "",
    98→    maxRadiusKm: 20,

   100→    passportPublic: true,
   101→  });
   102→
   103→  const load = useCallback(async () => {
   104→    try {
   105→      const res = await fetch("/api/worker/profile", { cache: "no-store" });
   106→      if (res.status === 403 || res.status === 404) {
   107→        setNotFound(true);
   108→        return;
   109→      }
   110→      if (!res.ok) return;
   111→      const data = (await res.json()) as WorkerProfileData;
   112→      setProfile(data);
   113→      setEdit({
   114→        fullName: data.fullName,
   115→        yearsExp: String(data.yearsExp),
   116→        city: data.city,
   117→        wageMin: String(data.wageMin),
   118→        wageMax: String(data.wageMax),
   119→        shiftPref: data.shiftPref,
   120→        bio: data.bio,
   121→        photoUrl: data.photoUrl ?? "",
   122→        maxRadiusKm: data.maxRadiusKm,
   123→        availableToday: data.availableToday,
   124→        passportPublic: data.passportPublic,
   125→      });
   126→    } catch {}
   127→  }, []);
   128→
   129→  useEffect(() => { load(); }, [load]);
   130→
   131→  // Live strength meter
   132→  const liveStrength = useMemo(() => {
   133→    if (!profile) return 0;
   134→    const has = (b: boolean) => (b ? 1 : 0);
   135→    const filled =
   136→      has(!!edit.fullName) +
   137→      has(Number(edit.yearsExp) > 0) +
   138→      has(!!edit.city) +
   139→      has(Number(edit.wageMin) > 0 || Number(edit.wageMax) > 0) +
   140→      has(!!edit.shiftPref) +
   141→      has(profile.languages.length > 0) +
   142→      has(!!edit.bio && edit.bio.length > 0);
   143→    let s = 30 + 10 * filled;
   144→    s += Math.min(25, 5 * profile.skills.length);
   145→    if (edit.bio && edit.bio.length > 50) s += 10;
   146→    if (edit.photoUrl) s += 10;
   147→    if (profile.trustTier !== "new") s += 10;
   148→    return Math.min(100, s);
   149→  }, [profile, edit]);
   150→
   151→  async function saveField(field: string, value: unknown) {
   152→    setSaving(true);
   153→    try {
   154→      const res = await fetch("/api/worker/profile", {
   155→        method: "PATCH",
   156→        headers: { "content-type": "application/json" },
   157→        body: JSON.stringify({ [field]: value }),
   158→      });
   159→      if (!res.ok) throw new Error("save-failed");
   160→      await load();
   161→      toast.success(t("save"));
   162→    } catch {
   163→      toast.error(t("errGeneric"));
   164→    } finally {
   165→      setSaving(false);
   166→    }
   167→  }
   168→
   169→  async function toggleAvailable(checked: boolean) {
   170→    setEdit(e => ({ ...e, availableToday: checked }));
   171→    setSaving(true);
   172→    try {
   173→      const res = await fetch("/api/worker/profile", {
   174→        method: "PATCH",
   175→        headers: { "content-type": "application/json" },
   176→        body: JSON.stringify({ availableToday: checked }),
   177→      });
   178→      if (!res.ok) throw new Error("save-failed");
   179→      toast.success(checked ? t("boardAvailableTodayOn") : t("boardAvailableTodayOff"));
   180→      await load();
   181→    } catch {
   182→      setEdit(e => ({ ...e, availableToday: !checked }));
   183→      toast.error(t("errGeneric"));
   184→    } finally {
   185→      setSaving(false);
   186→    }
   187→  }
   188→
   189→  async function togglePassportPublic(checked: boolean) {

   193→      const res = await fetch("/api/worker/profile", {
   194→        method: "PATCH",
   195→        headers: { "content-type": "application/json" },

   221→        maxRadiusKm: edit.maxRadiusKm,

   224→        body.lat = CITY_LATLNG[edit.city].lat;
   225→        body.lng = CITY_LATLNG[edit.city].lng;
   226→      }
   227→      const res = await fetch("/api/worker/profile", {
   228→        method: "PATCH",
   229→        headers: { "content-type": "application/json" },
   230→        body: JSON.stringify(body),
   231→      });
   232→      if (!res.ok) throw new Error("save-failed");
   233→      toast.success(t("save"));
   234→      await load();
   235→    } catch {
   236→      toast.error(t("errGeneric"));
   237→    } finally {
   238→      setSaving(false);
   239→    }
