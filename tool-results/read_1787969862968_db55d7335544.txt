    42→    42→function JobBoard() {
    43→    43→  const { t } = useLanguage();
    44→    44→  const router = useRouter();
    45→    45→  const search = useSearchParams();
    46→    46→  const { data: session, status } = useSession();
    47→    47→  const { savedIds, savedCount } = useSavedJobs();
    48→    48→
    49→    49→  // ---- URL-synced state (shareable deep links: /jobs?q=welder&city=...&sort=wage) ----
    50→    50→  const [query, setQuery] = useState(search.get("q") ?? "");
    51→    51→  const [tradeId, setTradeId] = useState(search.get("trade") ?? "");
    52→    52→  const [city, setCity] = useState(search.get("city") ?? "");
    53→    53→  const [shift, setShift] = useState<"day" | "night" | "any">((search.get("shift") as any) ?? "any");
    54→    54→  const [sort, setSort] = useState<SortKey>((search.get("sort") as SortKey) ?? "match");
    55→    55→  const [urgentOnly, setUrgentOnly] = useState(search.get("urgent") === "1");
    56→    56→  const [savedOnly, setSavedOnly] = useState(search.get("saved") === "1");

    58→    58→
    59→    59→  // ---- data ----
    60→    60→  const [skills, setSkills] = useState<Skill[]>([]);
    61→    61→  const [jobs, setJobs] = useState<WorkerJobCardData[] | null>(null);
    62→    62→  const [hasNext, setHasNext] = useState(false);
    63→    63→  const [loadingMore, setLoadingMore] = useState(false);
    64→    64→  const [profileExists, setProfileExists] = useState<boolean | null>(null);
    65→    65→  const [appliedIds, setAppliedIds] = useState<Set<string>>(new Set());
    66→    66→
    67→    67→  // ---- worker "Available today" quick-toggle (round 11) ----
    68→    68→  // Visible only to logged-in workers. Mirrors the toggle on /home but lets
    69→    69→  // a worker flip their availability while browsing the board.

    71→    71→  const [availableSaving, setAvailableSaving] = useState(false);

    80→    80→
    81→    81→  // Worker profile existence (redirect to onboarding when missing).
    82→    82→  // Non-workers (employer/admin) can browse the board too — no profile needed.
    83→    83→  useEffect(() => {
    84→    84→    if (status !== "authenticated") return;
    85→    85→    const role = (session?.user as { role?: string } | undefined)?.role;

    88→    88→      .then(r => {
    89→    89→        if (r.status === 403 || r.status === 404) { setProfileExists(false); return null; }
    90→    90→        if (!r.ok) return null;
    91→    91→        return r.json();
    92→    92→      })
    93→    93→      .then(data => {
    94→    94→        if (data === null) return;
    95→    95→        setProfileExists(true);
    96→    96→        // Round 11: seed the quick-toggle from the worker's stored flag.
    97→    97→        if (typeof data.availableToday === "boolean") {
    98→    98→          setAvailableToday(data.availableToday);
    99→    99→        }
   100→   100→      })
   101→   101→      .catch(() => setProfileExists(false));
   102→   102→  }, [status, session]);
   103→   103→
   104→   104→  // Round 11: workers can flip availability directly from the board header.
   105→   105→  async function toggleAvailableToday(checked: boolean) {
   106→   106→    if (availableToday === null) return;
   107→   107→    setAvailableToday(checked);
   108→   108→    setAvailableSaving(true);
   109→   109→    try {
   110→   110→      const res = await fetch("/api/worker/profile", {
   111→   111→        method: "PATCH",
   112→   112→        headers: { "content-type": "application/json" },
   113→   113→        body: JSON.stringify({ availableToday: checked }),
   114→   114→      });
   115→   115→      if (!res.ok) throw new Error("patch-failed");
   116→   116→      toast.success(checked ? t("boardAvailableTodayOn") : t("boardAvailableTodayOff"));
   117→   117→    } catch {
   118→   118→      setAvailableToday(!checked); // rollback
   119→   119→      toast.error(t("errGeneric"));
   120→   120→    } finally {
   121→   121→      setAvailableSaving(false);
   122→   122→    }
   123→   123→  }
   124→   124→
   125→   125→  useEffect(() => {
   126→   126→    if (profileExists === false && status === "authenticated") {
   127→   127→      const role = (session?.user as any)?.role;
   128→   128→      if (role === "worker") router.replace("/onboarding/worker");
   129→   129→    }
   130→   130→  }, [profileExists, status, session, router]);
   131→   131→
   132→   132→  // Applied jobs (mark cards so the Apply button doesn't double-submit)
   133→   133→  useEffect(() => {
   134→   134→    if (status !== "authenticated") return;
   135→   135→    fetch("/api/applications/mine", { cache: "no-store" })
   136→   136→      .then(r => (r.ok ? r.json() : { items: [] }))
   137→   137→      .then((d: { items: { jobId: string }[] }) => {
   138→   138→        setAppliedIds(new Set((d.items ?? []).map(a => a.jobId)));
   139→   139→      })
   140→   140→      .catch(() => {});
   141→   141→  }, [status]);
   142→   142→
   143→   143→  // Server-side query (trade/shift/urgent). Browse-all: distanceKm=200 overrides
   144→   144→  // the worker-radius filter so every open job on the platform is reachable here.
   145→   145→  const serverQuery = useMemo(() => {
   146→   146→    const p = new URLSearchParams();
   147→   147→    p.set("pageSize", String(PAGE_SIZE));
   148→   148→    p.set("distanceKm", "200");
   149→   149→    if (tradeId) p.set("tradeId", tradeId);
   150→   150→    if (shift !== "any") p.set("shift", shift);
   151→   151→    if (urgentOnly) p.set("urgentOnly", "true");
   152→   152→    return p.toString();
   153→   153→  }, [tradeId, shift, urgentOnly]);
   154→   154→
   155→   155→  // Fetch page 1 (debounced) whenever server-side filters change
   156→   156→  const loadFirstPage = useCallback(async () => {
   157→   157→    setJobs(null);
   158→   158→    try {
   159→   159→      const res = await fetch(`/api/jobs?${serverQuery}&page=1`, { cache: "no-store" });
   160→   160→      if (!res.ok) { setJobs([]); setHasNext(false); return; }
   161→   161→      const data = (await res.json()) as FeedResponse;
   162→   162→      setJobs(data.items ?? []);
   163→   163→      setHasNext(!!data.hasNext);
   164→   164→    } catch {
   165→   165→      setJobs([]);
   166→   166→      setHasNext(false);
   167→   167→    }
   168→   168→  }, [serverQuery]);
   169→   169→
   170→   170→  useEffect(() => {
   171→   171→    if (!profileExists) return;
   172→   172→    const id = setTimeout(loadFirstPage, 250);
   173→   173→    return () => clearTimeout(id);
   174→   174→  }, [loadFirstPage, profileExists]);
   175→   175→
   176→   176→  // Load-more appends the next page
   177→   177→  const loadMore = useCallback(async () => {
   178→   178→    if (loadingMore || !jobs) return;
   179→   179→    setLoadingMore(true);
   180→   180→    try {
   181→   181→      // Infer next page from the items we already hold
   182→   182→      const nextPage = Math.floor(jobs.length / PAGE_SIZE) + 1;
   183→   183→      const res = await fetch(`/api/jobs?${serverQuery}&page=${nextPage}`, { cache: "no-store" });
   184→   184→      if (!res.ok) { toast.error(t("errGeneric")); return; }
   185→   185→      const data = (await res.json()) as FeedResponse;
   186→   186→      setJobs(prev => {
   187→   187→        const seen = new Set((prev ?? []).map(j => j.id));
   188→   188→        const fresh = (data.items ?? []).filter(j => !seen.has(j.id));
   189→   189→        return [...(prev ?? []), ...fresh];
   190→   190→      });
   191→   191→      setHasNext(!!data.hasNext);
   192→   192→    } catch {
   193→   193→      toast.error(t("errGeneric"));
   194→   194→    } finally {
   195→   195→      setLoadingMore(false);
   196→   196→    }
   197→   197→  }, [loadingMore, jobs, serverQuery, t]);
   198→   198→
   199→   199→  // ---- URL sync (replace, no scroll jump) ----
   200→   200→  useEffect(() => {
   201→   201→    if (typeof window === "undefined") return;
   202→   202→    const p = new URLSearchParams();
   203→   203→    if (query.trim()) p.set("q", query.trim());
   204→   204→    if (tradeId) p.set("trade", tradeId);
   205→   205→    if (city) p.set("city", city);
   206→   206→    if (shift !== "any") p.set("shift", shift);
   207→   207→    if (sort !== "match") p.set("sort", sort);
   208→   208→    if (urgentOnly) p.set("urgent", "1");
   209→   209→    if (savedOnly) p.set("saved", "1");
   210→   210→    if (topEmployersOnly) p.set("top", "1");
   211→   211→    const qs = p.toString();
   212→   212→    const target = qs ? `/jobs?${qs}` : "/jobs";
   213→   213→    if (window.location.pathname + window.location.search !== target) {
   214→   214→      router.replace(target, { scroll: false });
   215→   215→    }
   216→   216→  }, [query, tradeId, city, shift, sort, urgentOnly, savedOnly, topEmployersOnly, router]);
   217→   217→
   218→   218→  const tradeSkills = useMemo(() => skills.filter(s => TRADE_NAMES.has(s.nameEn)), [skills]);
   219→   219→  const tradeName = useMemo(
   220→   220→    () => tradeSkills.find(s => s.id === tradeId)?.nameEn ?? null,
   221→   221→    [tradeSkills, tradeId],
   222→   222→  );
   223→   223→
   224→   224→  // Cities derived from loaded jobs (only cities that actually have jobs)
   225→   225→  const cities = useMemo(() => {
   226→   226→    if (!jobs) return [];
   227→   227→    return Array.from(new Set(jobs.map(j => j.city))).sort();
   228→   228→  }, [jobs]);
   229→   229→
   230→   230→  // ---- client-side pipeline: text search → city → saved → top employers → sort ----
   231→   231→  const visible = useMemo(() => {
   232→   232→    if (!jobs) return null;
   233→   233→    const q = query.trim().toLowerCase();
   234→   234→    let out = jobs.filter(j => {
   235→   235→      if (q) {
   236→   236→        const hay = [
   237→   237→          j.title,
   238→   238→          j.employer?.companyName ?? "",
   239→   239→          j.trade?.nameEn ?? "",
   240→   240→          j.city,
   241→   241→          ...j.skills.map(s => s.skill?.nameEn ?? ""),
   242→   242→        ].join(" ").toLowerCase();
   243→   243→        if (!hay.includes(q)) return false;
   244→   244→      }
   245→   245→      if (city && j.city !== city) return false;
   246→   246→      if (savedOnly && !savedIds.has(j.id)) return false;
   247→   247→      // Round 8: "Top employers" — jobs from employers rated ≥4.5 by 3+ workers
   248→   248→      if (topEmployersOnly) {
   249→   249→        const r = j.employer;
   250→   250→        if (!r?.ratingAvg || !r.ratingCount || r.ratingAvg < 4.5 || r.ratingCount < 3) return false;
   251→   251→      }
   252→   252→      return true;
   253→   253→    });
   254→   254→    out = [...out].sort((a, b) => {
   255→   255→      if (sort === "match") return (b.matchScore ?? -1) - (a.matchScore ?? -1);
   256→   256→      if (sort === "wage") return b.wageMax - a.wageMax;
   257→   257→      if (sort === "nearest") return (a.distanceKm ?? Infinity) - (b.distanceKm ?? Infinity);
   258→   258→      return +new Date(b.createdAt) - +new Date(a.createdAt);
   259→   259→    });
   260→   260→    return out;
   261→   261→  }, [jobs, query, city, savedOnly, savedIds, sort, topEmployersOnly]);
   262→   262→
   263→   263→  // ---- active filter chips ----
   264→   264→  interface Chip { key: string; label: string; clear: () => void }
   265→   265→  const chips: Chip[] = useMemo(() => {
   266→   266→    const out: Chip[] = [];
   267→   267→    if (query.trim()) out.push({ key: "q", label: `“${query.trim()}”`, clear: () => setQuery("") });
   268→   268→    if (tradeName) out.push({ key: "trade", label: tradeName, clear: () => setTradeId("") });
   269→   269→    if (city) out.push({ key: "city", label: city, clear: () => setCity("") });
   270→   270→    if (shift !== "any") out.push({ key: "shift", label: shift === "day" ? "Day" : "Night", clear: () => setShift("any") });
   271→   271→    if (urgentOnly) out.push({ key: "urgent", label: t("feedUrgent"), clear: () => setUrgentOnly(false) });
   272→   272→    if (savedOnly) out.push({ key: "saved", label: t("boardSavedOnly"), clear: () => setSavedOnly(false) });
   273→   273→    if (topEmployersOnly) out.push({ key: "top", label: t("boardTopEmployers"), clear: () => setTopEmployersOnly(false) });
   274→   274→    return out;
   275→   275→  }, [query, tradeName, city, shift, urgentOnly, savedOnly, topEmployersOnly, t]);

   277→   277→  function clearAll() {
   278→   278→    setQuery("");
   279→   279→    setTradeId("");
   280→   280→    setCity("");
   281→   281→    setShift("any");
   282→   282→    setUrgentOnly(false);
   283→   283→    setSavedOnly(false);
   284→   284→    setTopEmployersOnly(false);
   285→   285→    // keep sort — it's a view preference, not a filter
   286→   286→  }
   287→   287→
   288→   288→  const hasActiveFilters = chips.length > 0;
   289→   289→  const isWorker = (session?.user as { role?: string } | undefined)?.role === "worker";
   290→   290→
   291→   291→  if (status === "loading" || profileExists === null) {
   292→   292→    return (
   293→   293→      <AppShell>
   294→   294→        <LoadingSkeleton count={3} />
   295→   295→      </AppShell>
   296→   296→    );
   297→   297→  }
   298→   298→
   299→   299→  return (
   300→   300→    <AppShell>
   301→   301→      <div className="flex flex-col gap-4">
   302→   302→        {/* ---------- Hero / search header ---------- */}
   303→   303→        <section className="relative overflow-hidden rounded-2xl border border-border bg-gradient-to-br from-primary/10 via-card to-accent/10 p-4 sm:p-6">
   304→   304→          {/* decorative bridge arcs */}
   305→   305→          <div aria-hidden className="pointer-events-none absolute -right-10 -top-16 size-48 rounded-full border-[14px] border-primary/10" />
   306→   306→          <div aria-hidden className="pointer-events-none absolute -right-4 -top-10 size-28 rounded-full border-[10px] border-accent/15" />
   307→   307→
   308→   308→          <div className="relative flex flex-col gap-3">
   309→   309→            <div className="flex items-start justify-between gap-3 flex-wrap">
   310→   310→              <div>
   311→   311→                <h1 className="text-2xl font-bold tracking-tight flex items-center gap-2">
   312→   312→                  <Compass className="size-6 text-primary" />
   313→   313→                  {t("boardTitle")}
   314→   314→                </h1>
   315→   315→                <p className="text-sm text-muted-foreground mt-1">{t("boardSubtitle")}</p>
   316→   316→              </div>
   317→   317→              {visible && (
   318→   318→                <Badge variant="secondary" className="tabular-nums px-3 py-1 text-xs shrink-0">
   319→   319→                  {t(visible.length === 1 ? "boardResultOne" : "boardResults", { count: visible.length })}
   320→   320→                </Badge>
   321→   321→              )}
   322→   322→            </div>
   323→   323→
   324→   324→            {/* Search bar */}
   325→   325→            <div className="relative">
   326→   326→              <Search className="absolute left-3.5 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" aria-hidden />
   327→   327→              <Input
   328→   328→                type="search"
   329→   329→                value={query}
   330→   330→                onChange={e => setQuery(e.target.value)}
   331→   331→                placeholder={t("boardSearchPlaceholder")}
   332→   332→                aria-label={t("boardSearchPlaceholder")}
   333→   333→                className="pl-10 pr-10 h-12 text-base rounded-xl bg-card shadow-sm border-border/80 focus-visible:ring-2"
   334→   334→              />
   335→   335→              {query && (
   336→   336→                <button
   337→   337→                  type="button"
   338→   338→                  onClick={() => setQuery("")}
   339→   339→                  aria-label={t("boardClearAll")}
   340→   340→                  className="absolute right-2.5 top-1/2 -translate-y-1/2 grid place-items-center size-7 rounded-full text-muted-foreground hover:bg-accent hover:text-foreground transition-colors"
   341→   341→                >
   342→   342→                  <X className="size-4" />
   343→   343→                </button>
   344→   344→              )}
   345→   345→            </div>
   346→   346→
   347→   347→            {/* Round 11: worker "Available today" quick toggle */}
   348→   348→            {isWorker && availableToday !== null && (
   349→   349→              <motion.button
   350→   350→                type="button"
   351→   351→                onClick={() => toggleAvailableToday(!availableToday)}
   352→   352→                disabled={availableSaving}
   353→   353→                initial={{ opacity: 0, y: -4 }}
   354→   354→                animate={{ opacity: 1, y: 0 }}
   355→   355→                transition={{ duration: 0.25, ease: "easeOut" }}
   356→   356→                aria-pressed={availableToday}
   357→   357→                aria-label={t("boardAvailableToday")}
   358→   358→                className={`group inline-flex items-center gap-3 self-start rounded-xl border px-4 py-2.5 text-left transition-all duration-200 ${
   359→   359→                  availableToday
   360→   360→                    ? "border-emerald-400/70 bg-emerald-50 dark:bg-emerald-950/40 shadow-[0_0_0_3px_rgba(16,185,129,0.08)]"
   361→   361→                    : "border-border bg-card hover:border-emerald-300 hover:bg-emerald-50/40 dark:hover:bg-emerald-950/20"
   362→   362→                }`}
   363→   363→              >
   364→   364→                <span
   365→   365→                  aria-hidden
   366→   366→                  className={`relative grid size-9 place-items-center rounded-full transition-colors ${
   367→   367→                    availableToday

   369→   369→                      : "bg-muted text-muted-foreground group-hover:bg-emerald-100 group-hover:text-emerald-700"
   370→   370→                  }`}
   371→   371→                >
   372→   372→                  {availableSaving ? (
   373→   373→                    <Loader2 className="size-4 animate-spin" />
   374→   374→                  ) : (
   375→   375→                    <Radio className={`size-4 ${availableToday ? "animate-pulse" : ""}`} />

   377→   377→                  {availableToday && (

   379→   379→                  )}
   380→   380→                </span>
   381→   381→                <span className="flex flex-col gap-0.5 min-w-0">
   382→   382→                  <span className={`text-sm font-semibold leading-tight ${availableToday ? "text-emerald-800 dark:text-emerald-200" : "text-foreground"}`}>
   383→   383→                    {availableToday ? t("boardAvailableTodayOn") : t("boardAvailableTodayOff")}
   384→   384→                  </span>
   385→   385→                  <span className="text-[11px] text-muted-foreground truncate">

   387→   387→                  </span>
   388→   388→                </span>
   389→   389→                <Switch

   392→   392→                  onCheckedChange={toggleAvailableToday}
   393→   393→                  onClick={e => e.stopPropagation()} // avoid double-trigger from outer button
   394→   394→                  className="ml-1 data-[state=checked]:bg-emerald-600"
   395→   395→                  aria-label={t("boardAvailableToday")}
   396→   396→                />
   397→   397→              </motion.button>
   398→   398→            )}
   399→   399→          </div>
   400→   400→        </section>
   401→   401→
   402→   402→        {/* ---------- Sticky toolbar ---------- */}
