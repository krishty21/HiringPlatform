    87→export default function WorkerProfilePage() {
    88→  const { t, lang } = useLanguage();
    89→  const [profile, setProfile] = useState<WorkerProfileData | null>(null);
    90→  const [notFound, setNotFound] = useState(false);
    91→  const [saving, setSaving] = useState(false);
    92→  const [edit, setEdit] = useState({
    93→    fullName: "",
    94→    yearsExp: "0",
    95→    city: "",
    96→    wageMin: "0",
    97→    wageMax: "0",
    98→    shiftPref: "any" as "day" | "night" | "any",
    99→    bio: "",
   100→    photoUrl: "",
   101→    maxRadiusKm: 20,
   102→    availableToday: false,
   103→    passportPublic: true,
   104→  });
   105→
   106→  const load = useCallback(async () => {
   107→    try {
   108→      const res = await fetch("/api/worker/profile", { cache: "no-store" });
   109→      if (res.status === 403 || res.status === 404) {
   110→        setNotFound(true);
   111→        return;
   112→      }
   113→      if (!res.ok) return;
   114→      const data = (await res.json()) as WorkerProfileData;
   115→      setProfile(data);
   116→      setEdit({
   117→        fullName: data.fullName,
   118→        yearsExp: String(data.yearsExp),
   119→        city: data.city,
   120→        wageMin: String(data.wageMin),
   121→        wageMax: String(data.wageMax),
   122→        shiftPref: data.shiftPref,
   123→        bio: data.bio,
   124→        photoUrl: data.photoUrl ?? "",
   125→        maxRadiusKm: data.maxRadiusKm,
   126→        availableToday: data.availableToday,
   127→        passportPublic: data.passportPublic,
   128→      });
   129→    } catch {}
   130→  }, []);
   131→
   132→  useEffect(() => { load(); }, [load]);
   133→
   134→  // Live strength meter — computed from the editable form state + server-side
   135→  // languages/skills/trust data.
   136→  const liveStrength = useMemo(() => {
   137→    if (!profile) return 0;
   138→    const has = (b: boolean) => (b ? 1 : 0);
   139→    const filled =
   140→      has(!!edit.fullName) +
   141→      has(Number(edit.yearsExp) > 0) +
   142→      has(!!edit.city) +
   143→      has(Number(edit.wageMin) > 0 || Number(edit.wageMax) > 0) +
   144→      has(!!edit.shiftPref) +
   145→      has(profile.languages.length > 0) +
   146→      has(!!edit.bio && edit.bio.length > 0);
   147→    let s = 30 + 10 * filled;
   148→    s += Math.min(25, 5 * profile.skills.length);
   149→    if (edit.bio && edit.bio.length > 50) s += 10;
   150→    if (edit.photoUrl) s += 10;
   151→    if (profile.trustTier !== "new") s += 10;
   152→    return Math.min(100, s);
   153→  }, [profile, edit]);
   154→
   155→  async function saveField(field: string, value: unknown) {
   156→    setSaving(true);
   157→    try {
   158→      const res = await fetch("/api/worker/profile", {
   159→        method: "PATCH",
   160→        headers: { "content-type": "application/json" },
   161→        body: JSON.stringify({ [field]: value }),
   162→      });
   163→      if (!res.ok) throw new Error("save-failed");
   164→      await load();
   165→      toast.success(t("save"));
   166→    } catch {
   167→      toast.error(t("errGeneric"));
   168→    } finally {
   169→      setSaving(false);
   170→    }
   171→  }
   172→
   173→  async function toggleAvailable(checked: boolean) {
   174→    setEdit(e => ({ ...e, availableToday: checked }));
   175→    setSaving(true);
   176→    try {
   177→      const res = await fetch("/api/worker/profile", {
   178→        method: "PATCH",
   179→        headers: { "content-type": "application/json" },
   180→        body: JSON.stringify({ availableToday: checked }),
   181→      });
   182→      if (!res.ok) throw new Error("save-failed");
   183→      toast.success(checked ? t("boardAvailableTodayOn") : t("boardAvailableTodayOff"));
   184→      await load();
   185→    } catch {
   186→      setEdit(e => ({ ...e, availableToday: !checked }));
   187→      toast.error(t("errGeneric"));
   188→    } finally {
   189→      setSaving(false);
   190→    }
   191→  }
   192→
   193→  async function togglePassportPublic(checked: boolean) {
   194→    setEdit(e => ({ ...e, passportPublic: checked }));
   195→    setSaving(true);
   196→    try {
   197→      const res = await fetch("/api/worker/profile", {
   198→        method: "PATCH",
   199→        headers: { "content-type": "application/json" },
   200→        body: JSON.stringify({ passportPublic: checked }),
   201→      });
   202→      if (!res.ok) throw new Error("save-failed");
   203→      toast.success(t("save"));
   204→      await load();
   205→    } catch {
   206→      setEdit(e => ({ ...e, passportPublic: !checked }));
   207→      toast.error(t("errGeneric"));
   208→    } finally {
   209→      setSaving(false);
   210→    }
   211→  }
   212→
   213→  async function saveProfileEdits() {
   214→    setSaving(true);
   215→    try {
   216→      const body: Record<string, unknown> = {
   217→        fullName: edit.fullName,
   218→        yearsExp: Number(edit.yearsExp) || 0,
   219→        city: edit.city,
   220→        wageMin: Number(edit.wageMin) || 0,
   221→        wageMax: Number(edit.wageMax) || 0,
   222→        shiftPref: edit.shiftPref,
   223→        bio: edit.bio,
   224→        photoUrl: edit.photoUrl || null,
   225→        maxRadiusKm: edit.maxRadiusKm,
   226→      };
   227→      if (CITY_LATLNG[edit.city]) {
   228→        body.lat = CITY_LATLNG[edit.city].lat;
   229→        body.lng = CITY_LATLNG[edit.city].lng;
   230→      }
   231→      const res = await fetch("/api/worker/profile", {
   232→        method: "PATCH",
   233→        headers: { "content-type": "application/json" },
   234→        body: JSON.stringify(body),
   235→      });
   236→      if (!res.ok) throw new Error("save-failed");
   237→      toast.success(t("save"));
   238→      await load();
   239→    } catch {
   240→      toast.error(t("errGeneric"));
   241→    } finally {
   242→      setSaving(false);
   243→    }
   244→  }

   249→      ? (lang === "hi" ? profile.trade.nameHi : lang === "te" ? profile.trade.nameTe : profile.trade.nameEn)

   252→    const url = `https://wa.me/?text=${encodeURIComponent(text)}`;
   253→    window.open(url, "_blank", "noopener,noreferrer");
   254→  }
   255→
   256→  if (notFound) {

   319→            <div className="flex-1 min-w-0 flex flex-col gap-1.5">

   357→                {t("passportTier")}

   417→          {/* Identity / Skills / Reliability — three-layer dl */}

   423→                  {t("passportTierIdVerified")}
