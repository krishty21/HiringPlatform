    83→export default function WorkerProfilePage() {
    84→  const { t, lang } = useLanguage();
    85→  const [profile, setProfile] = useState<WorkerProfileData | null>(null);
    86→  const [notFound, setNotFound] = useState(false);
    87→  const [saving, setSaving] = useState(false);
    88→  const [edit, setEdit] = useState({
    89→    fullName: "",
    90→    yearsExp: "0",
    91→    city: "",
    92→    wageMin: "0",
    93→    wageMax: "0",
    94→    shiftPref: "any" as "day" | "night" | "any",
    95→    bio: "",
    96→    photoUrl: "",
    97→    maxRadiusKm: 20,
    98→    availableToday: false,
    99→    passportPublic: true,
   100→  });
   101→
   102→  const load = useCallback(async () => {
   103→    try {
   104→      const res = await fetch("/api/worker/profile", { cache: "no-store" });
   105→      if (res.status === 403 || res.status === 404) {
   106→        setNotFound(true);
   107→        return;
   108→      }
   109→      if (!res.ok) return;
   110→      const data = (await res.json()) as WorkerProfileData;
   111→      setProfile(data);
   112→      setEdit({
   113→        fullName: data.fullName,
   114→        yearsExp: String(data.yearsExp),
   115→        city: data.city,
   116→        wageMin: String(data.wageMin),
   117→        wageMax: String(data.wageMax),
   118→        shiftPref: data.shiftPref,
   119→        bio: data.bio,
   120→        photoUrl: data.photoUrl ?? "",
   121→        maxRadiusKm: data.maxRadiusKm,
   122→        availableToday: data.availableToday,
   123→        passportPublic: data.passportPublic,
   124→      });
   125→    } catch {}
   126→  }, []);
   127→
   128→  useEffect(() => { load(); }, [load]);
   129→
   130→  // Live strength meter
   131→  const liveStrength = useMemo(() => {
   132→    if (!profile) return 0;
   133→    const has = (b: boolean) => (b ? 1 : 0);
   134→    const filled =
   135→      has(!!edit.fullName) +
   136→      has(Number(edit.yearsExp) > 0) +
   137→      has(!!edit.city) +
   138→      has(Number(edit.wageMin) > 0 || Number(edit.wageMax) > 0) +
   139→      has(!!edit.shiftPref) +
   140→      has(profile.languages.length > 0) +
   141→      has(!!edit.bio && edit.bio.length > 0);
   142→    let s = 30 + 10 * filled;
   143→    s += Math.min(25, 5 * profile.skills.length);
   144→    if (edit.bio && edit.bio.length > 50) s += 10;
   145→    if (edit.photoUrl) s += 10;
   146→    if (profile.trustTier !== "new") s += 10;
   147→    return Math.min(100, s);
   148→  }, [profile, edit]);
   149→
   150→  async function saveField(field: string, value: unknown) {
   151→    setSaving(true);
   152→    try {
   153→      const res = await fetch("/api/worker/profile", {
   154→        method: "PATCH",
   155→        headers: { "content-type": "application/json" },
   156→        body: JSON.stringify({ [field]: value }),
   157→      });
   158→      if (!res.ok) throw new Error("save-failed");
   159→      await load();
   160→      toast.success(t("save"));
   161→    } catch {
   162→      toast.error(t("errGeneric"));
   163→    } finally {
   164→      setSaving(false);
   165→    }
   166→  }
   167→
   168→  async function toggleAvailable(checked: boolean) {
   169→    setEdit(e => ({ ...e, availableToday: checked }));
   170→    setSaving(true);
   171→    try {
   172→      const res = await fetch("/api/worker/profile", {
   173→        method: "PATCH",
   174→        headers: { "content-type": "application/json" },
   175→        body: JSON.stringify({ availableToday: checked }),
   176→      });
   177→      if (!res.ok) throw new Error("save-failed");
   178→      toast.success(checked ? "Available today on" : "Available today off");
   179→      await load();
   180→    } catch {
   181→      setEdit(e => ({ ...e, availableToday: !checked }));
   182→      toast.error(t("errGeneric"));
   183→    } finally {
   184→      setSaving(false);
   185→    }
   186→  }
   187→
   188→  async function togglePassportPublic(checked: boolean) {
   189→    setEdit(e => ({ ...e, passportPublic: checked }));
   190→    setSaving(true);
   191→    try {
   192→      const res = await fetch("/api/worker/profile", {
   193→        method: "PATCH",
   194→        headers: { "content-type": "application/json" },
   195→        body: JSON.stringify({ passportPublic: checked }),
   196→      });
   197→      if (!res.ok) throw new Error("save-failed");
   198→      toast.success(t("save"));
   199→      await load();
   200→    } catch {
   201→      setEdit(e => ({ ...e, passportPublic: !checked }));
   202→      toast.error(t("errGeneric"));
   203→    } finally {
   204→      setSaving(false);
   205→    }
   206→  }
   207→
   208→  async function saveProfileEdits() {
   209→    setSaving(true);
   210→    try {
   211→      const body: Record<string, unknown> = {
   212→        fullName: edit.fullName,
   213→        yearsExp: Number(edit.yearsExp) || 0,
   214→        city: edit.city,
   215→        wageMin: Number(edit.wageMin) || 0,
   216→        wageMax: Number(edit.wageMax) || 0,
   217→        shiftPref: edit.shiftPref,
   218→        bio: edit.bio,
   219→        photoUrl: edit.photoUrl || null,
   220→        maxRadiusKm: edit.maxRadiusKm,
   221→      };
   222→      if (CITY_LATLNG[edit.city]) {
   223→        body.lat = CITY_LATLNG[edit.city].lat;
   224→        body.lng = CITY_LATLNG[edit.city].lng;
   225→      }
   226→      const res = await fetch("/api/worker/profile", {
   227→        method: "PATCH",
   228→        headers: { "content-type": "application/json" },
   229→        body: JSON.stringify(body),
   230→      });
   231→      if (!res.ok) throw new Error("save-failed");
   232→      toast.success(t("save"));
   233→      await load();
   234→    } catch {
   235→      toast.error(t("errGeneric"));
   236→    } finally {
   237→      setSaving(false);
   238→    }
   239→  }

   241→  function shareWhatsApp() {
   242→    if (!profile) return;
   243→    const tradeName = profile.trade
   244→      ? (lang === "hi" ? profile.trade.nameHi : lang === "te" ? profile.trade.nameTe : profile.trade.nameEn)
   245→      : "Worker";

   247→    const url = `https://wa.me/?text=${encodeURIComponent(text)}`;
   248→    window.open(url, "_blank", "noopener,noreferrer");
   249→  }
   250→
   251→  if (notFound) {
   252→    return (
   253→      <AppShell>
   254→        <EmptyState
   255→          icon={ShieldCheck}

   257→          description={t("onboardTitle")}

   259→        />
   260→      </AppShell>
   261→    );
   262→  }
   263→
   264→  if (!profile) {

   267→        <LoadingSkeleton count={4} />

   270→  }
   271→

   290→      <div className="grid gap-4 lg:grid-cols-[1fr_320px]">
   291→        {/* Skill Passport card */}
   292→        <Card className="passport-card">
   293→          <CardContent className="p-6 flex flex-col gap-4">
   294→            <div className="flex items-start gap-4">
   295→              <Avatar className="size-16 border-2 border-primary/30">
   296→                {profile.photoUrl ? (
   297→                  <img src={profile.photoUrl} alt={profile.fullName} className="size-full object-cover" />
   298→                ) : (
   299→                  <AvatarFallback className="text-lg font-bold bg-primary/5 text-primary">{initials}</AvatarFallback>
   300→                )}
   301→              </Avatar>
   302→              <div className="flex-1">
   303→                <div className="flex items-start justify-between gap-2 flex-wrap">
   304→                  <div>
   305→                    <h2 className="text-xl font-bold tracking-tight">{profile.fullName}</h2>
   306→                    <p className="text-sm text-muted-foreground flex items-center gap-1.5 mt-1">
   307→                      <Briefcase className="size-3.5" />
   308→                      {tradeName} · {profile.yearsExp} {t("passportYears")}
   309→                    </p>
   310→                  </div>
   311→                  <TrustTierBadge tier={profile.trustTier} score={profile.trustScore} size="lg" />
   312→                </div>
   313→                <div className="flex flex-wrap items-center gap-2 mt-2">
   314→                  {profile.availableToday && (
   315→                    <Badge className="bg-emerald-100 text-emerald-800 border border-emerald-300 gap-1 text-xs">
   316→                      <Zap className="size-3" />
   317→                      {t("today")}
   318→                    </Badge>
   319→                  )}
   320→                  <Badge variant="outline" className="text-xs gap-1">
   321→                    <MapPin className="size-3" />
   322→                    {profile.city}
   323→                  </Badge>
   324→                  <Badge variant="outline" className="text-xs gap-1">
   325→                    <Eye className="size-3" />
   326→                    {profile.profileViews} views
   327→                  </Badge>
   328→                </div>
   329→              </div>
   330→            </div>
   331→
   332→            <Separator />
   333→
   334→            {/* Trust tier ladder — New → ID Verified → Skill Verified → Top Pro */}
   335→            <div className="grid gap-2 rounded-lg border border-border bg-card/50 p-3" aria-label={`${t("passportTier")}: ${profile.trustScore}/100`}>
   336→              <div className="flex items-center justify-between text-xs">
   337→                <span className="font-semibold flex items-center gap-1.5">
   338→                  <ShieldCheck className="size-3.5 text-primary" />
   339→                  {t("passportTier")}
   340→                </span>
   341→                <span className="tabular-nums font-bold">{profile.trustScore}/100 · {t("passportTrustScore")}</span>
   342→              </div>
   343→              <ol className="flex items-center gap-0" aria-hidden>
   344→                {([
   345→                  ["new", t("passportTierNew")],
   346→                  ["id_verified", t("passportTierIdVerified")],
   347→                  ["skill_verified", t("passportTierSkillVerified")],
   348→                  ["top_pro", t("passportTierTopPro")],
   349→                ] as const).map(([tier, label], i) => {
   350→                  const currentIdx = TRUST_LADDER.indexOf(profile.trustTier);
   351→                  const state = i < currentIdx ? "done" : i === currentIdx ? "current" : "todo";
   352→                  return (
   353→                    <li key={tier} className="flex-1 flex flex-col items-center gap-1.5 last:flex-none">
   354→                      <div className="flex items-center w-full">
   355→                        {i > 0 && (
   356→                          <span className={`h-0.5 flex-1 mx-0.5 rounded-full ${i <= currentIdx ? "bg-primary/60" : "bg-border"}`} />
   357→                        )}
   358→                        <span
   359→                          className={`size-6 rounded-full grid place-items-center border-2 text-[10px] font-bold transition-colors ${
   360→                            state === "done"
   361→                              ? "bg-primary border-primary text-primary-foreground"
   362→                              : state === "current"
   363→                                ? "bg-accent/15 border-accent text-accent-foreground ring-4 ring-accent/15"
   364→                                : "bg-muted/50 border-border text-muted-foreground/60"
   365→                          }`}
   366→                        >
   367→                          {state === "done" ? <Check className="size-3" /> : i + 1}
   368→                        </span>
   369→                        {i < 3 && (
   370→                          <span className={`h-0.5 flex-1 mx-0.5 rounded-full ${i < currentIdx ? "bg-primary/60" : "bg-border"}`} />
   371→                        )}
   372→                      </div>
   373→                      <span
   374→                        className={`text-[10px] font-medium text-center leading-tight ${
   375→                          state === "current" ? "text-accent-foreground" : "text-muted-foreground"
   376→                        } ${i === 3 ? "whitespace-nowrap" : ""}`}
   377→                      >
   378→                        {label}
   379→                      </span>
   380→                    </li>
   381→                  );
   382→                })}
   383→              </ol>
   384→            </div>
   385→
   386→            {/* Strength meter — live (WRK-04) */}
   387→            <div className="grid gap-2 rounded-lg border border-primary/20 bg-primary/5 p-3">
   388→              <div className="flex items-center justify-between text-xs">
   389→                <span className="font-semibold flex items-center gap-1.5">
   390→                  <Sparkles className="size-3.5" />
   391→                  {t("passportStrength")}
   392→                </span>
   393→                <span className="tabular-nums">{liveStrength}%</span>
   394→              </div>
   395→              <Progress value={liveStrength} className="h-2" />
   396→            </div>
   397→
   398→            <Separator />
   399→
   400→            {/* Editable fields */}
   401→            <div className="grid sm:grid-cols-2 gap-4">
   402→              <div className="grid gap-2">

   406→                  onChange={e => setEdit(s => ({ ...s, fullName: e.target.value }))}
   407→                  className="min-h-11"
   408→                />
   409→              </div>
   410→              <div className="grid gap-2">
   411→                <Label htmlFor="yearsExp">{t("onboardYearsExp")}</Label>

   414→                  value={edit.yearsExp}
   415→                  onChange={e => setEdit(s => ({ ...s, yearsExp: e.target.value }))}

   417→                />
   418→              </div>
   419→              <div className="grid gap-2">
   420→                <Label htmlFor="city">{t("onboardCity")}</Label>
   421→                <Select
   422→                  value={edit.city}
   423→                  onValueChange={(v) => setEdit(s => ({ ...s, city: v }))}
   424→                >
   425→                  <SelectTrigger id="city" className="min-h-11 w-full"><SelectValue /></SelectTrigger>

   429→                </Select>
   430→              </div>
   431→              <div className="grid gap-2">
   432→                <Label htmlFor="shift">{t("onboardShift")}</Label>
   433→                <Select
   434→                  value={edit.shiftPref}
   435→                  onValueChange={(v) => setEdit(s => ({ ...s, shiftPref: v as typeof edit.shiftPref }))}
   436→                >
   437→                  <SelectTrigger id="shift" className="min-h-11 w-full"><SelectValue /></SelectTrigger>
   438→                  <SelectContent>
   439→                    <SelectItem value="any">Any</SelectItem>
   440→                    <SelectItem value="day">Day</SelectItem>

   442→                  </SelectContent>
   443→                </Select>
   444→              </div>
   445→              <div className="grid gap-2">
   446→                <Label htmlFor="wageMin">{t("onboardWageMin")} (₹/day)</Label>

   454→              <div className="grid gap-2">

   470→                />
   471→              </div>
